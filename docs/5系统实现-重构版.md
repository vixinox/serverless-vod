```typescript
import { headers } from "next/headers";
import { auth } from "@/lib/auth";

export async function getSessionFromHeaders(requestHeaders?: Headers) {
  const resolvedHeaders = requestHeaders ?? await headers();
  return auth.api.getSession({ headers: resolvedHeaders });
}

export async function requireUserId(requestHeaders?: Headers) {
  const session = await getSessionFromHeaders(requestHeaders).catch(() => null);
  const userId = session?.user?.id ?? null;

}
```

```typescript
async function getVideosByType(type: "LONG" | "SHORT", cursor: string | null, limit: number) {
  const videos = await prisma.video.findMany({
    where: {
      deletedAt: null,
      visibility: "PUBLIC",
      processingStatus: "READY", // 仅查询已完成转码的视频
      type,
    },
    ...(cursor ? { cursor: { id: cursor }, skip: 1 } : {}),
    take: limit + 1, // 多查一条用于判断下一页
    orderBy: [{ createdAt: "desc" }, { id: "desc" }],
  });
  
  const hasMore = videos.length > limit;
  const page = hasMore ? videos.slice(0, limit) : videos;
  return {
    items: page,
    nextCursor: hasMore ? page[page.length - 1].id : null,
  };
}
```
查询条件中强制指定了 `visibility = PUBLIC` 与 `processingStatus = READY`，确保草稿和未转码完成的视频不会被越权读取。`take: limit + 1` 的逻辑能够在一次数据库查询中同时返回当前页数据与是否有下一页的游标标识。

```sql
CREATE OR REPLACE FUNCTION fn_comment_counter()
RETURNS TRIGGER LANGUAGE plpgsql AS $$
BEGIN
  IF TG_OP = 'INSERT' THEN
    UPDATE "Video" SET "commentsCount" = "commentsCount" + 1 WHERE "id" = NEW."videoId";
    IF NEW."parentId" IS NOT NULL THEN
      UPDATE "Comment" SET "repliesCount" = "repliesCount" + 1 WHERE "id" = NEW."parentId";
    END IF;
    RETURN NEW;
  ELSIF TG_OP = 'DELETE' THEN
    UPDATE "Video" SET "commentsCount" = GREATEST(0, "commentsCount" - 1) WHERE "id" = OLD."videoId";
    -- 省略部分层级更新逻辑
    RETURN OLD;
  END IF;
END;
$$;

CREATE TRIGGER trg_comment_counter
AFTER INSERT OR DELETE ON "Comment"
FOR EACH ROW EXECUTE FUNCTION fn_comment_counter();
```

> 代码来源：`scripts/localstack/lambda/transcode/index.mjs`
> 建议截取：论文正文建议优先使用第 347-361 行的 `transcodeSingleRenditionReencode`，因为这里最能体现真实转码参数；若要完整反映项目实现，还应同时补充第 364-378 行的 `transcodeSingleRenditionCopy` 与第 381-389 行的回退逻辑。需要注意：当前文档示例参数与项目真实实现并不完全一致，尤其是 `-profile:v main -level 4.0` 并未出现在现有代码中，正文最好按真实代码重写。
```bash
ffmpeg -y -v warning -i source.mp4 \
  -c:v libx264 -preset veryfast \
  -profile:v main -level 4.0 \
  -g 48 -sc_threshold 0 \
  -c:a aac -ar 48000 -b:a 128k \
  -hls_time 4 -hls_playlist_type vod \
  -hls_flags independent_segments \
  -hls_segment_filename out/seg_%03d.ts \
  out/master.m3u8
```
**核心参数解析：**
*   `-profile:v main -level 4.0`：限制编码配置为 Main Profile，以保证视频在移动端设备上能够正常进行硬件解码。
*   `-hls_time 4`：将 TS 分片时长固定为 4 秒，较短的分片可以加快播放器的首屏加载速度，并减少拖拽进度条时的缓冲时间。
*   `-hls_playlist_type vod`：指定生成点播型（VOD）播放列表。
*   `-g 48 -sc_threshold 0`：固定关键帧间隔（GOP），并关闭场景切换检测，这对于确保 HLS 切片边界的时间戳精确对齐至关重要。

系统通过定期执行 PostgreSQL 存储过程 `sp_rollup_daily_stats`，将原始播放日志和互动数据聚合为日统计表 `VideoDailyStat`。核心的汇总 SQL 逻辑如下：

> 代码来源：`prisma/rollup.daily.sql`
> 建议截取：第 24-86 行，即视频日统计的核心 `INSERT ... ON CONFLICT DO UPDATE` 逻辑；若希望明确这段 SQL 处于存储过程内部，可从第 7 行的 `CREATE OR REPLACE FUNCTION sp_rollup_daily_stats` 一直截到第 86 行。项目真实实现除播放量、独立访客和观看时长外，还额外汇总了点赞、点踩与评论增长量。
```sql
INSERT INTO "VideoDailyStat" ("videoId", "date", "views", "uniqueViewers", "watchTimeSeconds")
SELECT
  vpe."videoId",
  DATE(vpe."createdAt") AS stat_date,
  COUNT(*) FILTER (WHERE vpe."eventType" = 'PLAY_START') AS daily_views,
  COUNT(DISTINCT COALESCE(vpe."userId", vpe."sessionId")) AS daily_unique_viewers,
  FLOOR(SUM(COALESCE(vpe."watchDeltaMs", 0)) / 1000.0) AS daily_watch_time
FROM "VideoPlaybackEvent" vpe
WHERE DATE(vpe."createdAt") >= CURRENT_DATE - INTERVAL '1 day'
GROUP BY vpe."videoId", DATE(vpe."createdAt")
ON CONFLICT ("videoId", "date") DO UPDATE
SET
  "views" = EXCLUDED."views",
  "uniqueViewers" = EXCLUDED."uniqueViewers",
  "watchTimeSeconds" = EXCLUDED."watchTimeSeconds";
```
此实现的工程优势在于：
1. 使用 `COUNT(DISTINCT)` 结合 `userId` 和 `sessionId`，实现了登录用户与游客的独立访客（UV）去重统计。
2. 利用 `ON CONFLICT DO UPDATE`（Upsert 机制），保证了统计脚本重复执行时的幂等性，避免数据重复累加。
前端统计页面只需按时间范围读取 `VideoDailyStat` 这张轻量级的聚合表，即可快速渲染折线图和数据看板，大幅降低了数据库的实时查询负载。
